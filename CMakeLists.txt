cmake_minimum_required(VERSION 3.20)
project(VoidCamera LANGUAGES C CXX ASM_MASM)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "x64 required")
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin/v1.48")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/bin/v1.48/Release")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
set(D "${CMAKE_SOURCE_DIR}/third_party")
add_library(minhook STATIC ${D}/minhook/src/buffer.c ${D}/minhook/src/hook.c ${D}/minhook/src/trampoline.c ${D}/minhook/src/hde/hde64.c)
target_include_directories(minhook PUBLIC ${D}/minhook/include)
add_library(VoidCamera SHARED src/camera.cpp src/camera_hook.asm src/overlay.cpp ${D}/imgui/imgui.cpp ${D}/imgui/imgui_draw.cpp ${D}/imgui/imgui_tables.cpp ${D}/imgui/imgui_widgets.cpp ${D}/imgui/backends/imgui_impl_dx11.cpp ${D}/imgui/backends/imgui_impl_win32.cpp)
target_include_directories(VoidCamera PRIVATE ${D}/imgui)
target_link_libraries(VoidCamera PRIVATE minhook d3d11 dxgi user32 psapi Xinput9_1_0)
add_executable(VoidCameraInjector src/injector.cpp)
target_link_libraries(VoidCameraInjector PRIVATE bcrypt)
